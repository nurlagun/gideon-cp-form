/** * Add this to your Apps Script project (next to your existing doGet). * It accepts the new POST from the form, ignores repeated submissions * that carry the same submissionId, and reuses your existing doGet * "submit" logic so nothing else needs to change. * * After adding: Deploy > Manage deployments > Edit > Version: "New version" > Deploy * (keep the same /exec URL). */ function doPost(e) { const body = JSON.parse(e.postData.contents); const data = body.data || {}; const id = data.submissionId; const cache = CacheService.getScriptCache(); const lock = LockService.getScriptLock(); lock.waitLock(30000); // only one submission processed at a time try { // Already processed this exact booking? Return the original result, do nothing else. const previous = id && cache.get('sub_' + id); if (previous) { return ContentService.createTextOutput(previous) .setMimeType(ContentService.MimeType.JSON); } // Run your existing submit code (row insert + WhatsApp alerts). const out = doGet({ parameter: { action: 'submit', data: JSON.stringify(data) } }); const text = out.getContent(); // Remember successful submissions for 6 hours. try { if (id && JSON.parse(text).ok) cache.put('sub_' + id, text, 21600); } catch (err) {} return out; } finally { lock.releaseLock(); } }
